# openmentor.io — AI-crawler policy: docs/migration/DECISIONS.md (D35).
#
# Private pages (mentor portal, admin, contact forms) are deliberately NOT
# disallowed here: they carry <meta name="robots" content="noindex">, and a
# crawler blocked by robots.txt can never fetch the page to see that tag —
# which is how blocked URLs end up indexed as bare links anyway. Only the JSON
# API, which cannot carry a meta tag, is blocked.

User-agent: *
# Social-card images must stay fetchable: Facebook/LinkedIn respect
# robots.txt when loading og:image (longest-match Allow wins over /api/).
Allow: /api/og/
Disallow: /api/
Allow: /
